#Decimal to Binary conversion
def dec_bin(decimal):
    binary = 0
    if decimal >= 128:
        binary = binary + 10000000
        decimal = decimal - 128
    if decimal >= 64:
        binary = binary + 1000000
        decimal = decimal - 64
    if decimal >= 32:
        binary = binary + 100000
        decimal = decimal - 32
    if decimal >= 16:
        binary = binary + 10000
        decimal = decimal - 16
    if decimal >= 8:
        binary = binary + 1000
        decimal = decimal - 8
    if decimal >= 4:
        binary = binary + 100
        decimal = decimal - 4
    if decimal >= 2:
        binary = binary + 10
        decimal = decimal - 2
    if decimal >= 1:
        binary = binary + 1
    return binary

#Binary to Decimal conversion
def bin_dec(binary):
    decimal = 0
    if binary >= 10000000:
        decimal = decimal + 128
        binary = binary - 10000000
    if binary >= 1000000:
        decimal = decimal + 64
        binary = binary - 1000000
    if binary >= 100000:
        decimal = decimal + 32
        binary = binary - 100000
    if binary >= 10000:
        decimal = decimal + 16
        binary = binary - 10000   
    if binary >= 1000:
        decimal = decimal + 8
        binary = binary - 1000
    if binary >= 100:
        decimal = decimal + 4
        binary = binary - 100
    if binary >= 10:
        decimal = decimal + 2
        binary = binary -10
    if binary >= 1:
        decimal = decimal + 1
    return decimal

#Hex to Decimal
def hex_dec(hexa):
    decimal = 0
    for digit in str(hexa):
        if digit in my_list:
            value = 10
        
            for letter in my_list:
                if digit == letter:
                    break
                value = value + 1 
        else:
            value = int(digit)
        decimal = decimal * 16 + value
    return decimal
    
#Decimal to Hex
def dec_hex(decimal):
    hex_chars = "0123456789ABCDEF"
    first_num = decimal//16
    second_num = decimal%16
    return hex_chars[first_num] + hex_chars[second_num]
    
    return first_num + second_num

#Hex to Binary
def hex_bin(hexa):
    decimal_val = hex_dec(hexa)
    return dec_bin(decimal_val)

#Binary to Hex
def bin_hex(binary):
    decimal_val = bin_dec(binary)
    return dec_hex(decimal_val)
