def dec_to_bin(dec):
    binary_output = ""
    for i in range(8,0,-1):
        if dec - (2 ** (i-1)) >= 0 and dec != 0:
            binary_output += "1"
            dec = dec - (2 ** (i-1))
        else:
            binary_output += "0"
    return binary_output
    
def bin_to_dec(bin):
    bin = int(bin)
    decimal_output = 0
    for i in range(8,0,-1):
        if bin >= 10 ** (i-1):
            bin -= 10 ** (i-1)
            decimal_output += 2 ** (i-1)
    return decimal_output
