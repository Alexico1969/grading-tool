#Converting to Binary 
def binarycon(x):
    finnum = ""
    binarydiv = [128,64,32,16,8,4,2,1]
    for i in range(0,8):
        if x - binarydiv[i] >= 0:
            finnum += "1"
            x = x - binarydiv[i]
        else:
            finnum += "0"
    return finnum

#Converting to Decimal
